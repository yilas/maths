\chapter{Complément : construire exponentielle et logarithme}
\section*{Exercice 26.1 -- Approximer e}
Quelle borne d'erreur donne la somme $\sum_{k=0}^7 1/k!$ pour approcher $e$ ?

\section*{Exercice 26.2 -- Convergence uniforme}
Pourquoi $f_n(x)=x^n$ ne converge-t-elle pas uniformément sur $[0,1]$ ?

\section*{Exercice 26.3 -- Logarithmes}
Résoudre $\ln(x-1)+\ln(x+1)=\ln8$.

\section*{Exercice 26.4 -- Dérivée d'une inverse}
Retrouver la dérivée de $\ln$ en utilisant $h=e^{x+k}-e^x$ avec $x=\ln y$.

\section*{Exercice 26.5 -- Limite d'Euler}
À partir de $t/(1+t)\le\ln(1+t)\le t$ pour $t\ge0$, démontrer $(1+1/n)^n\to e$.

\section*{Exercice 26.6 -- Logarithme et aire}
Démontrer $\int_1^xdt/t=\ln x$ pour $x>0$, puis en déduire $\ln2>1/2$.

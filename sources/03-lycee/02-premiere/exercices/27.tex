\chapter{Complément : espaces vectoriels et projections}
\section*{Exercice 27.1 -- Noyau et image}
Pour $T(x,y)=(x-y,2x-2y)$, trouver noyau et image.

\section*{Exercice 27.2 -- Projection sur une droite}
Projeter $x=(2,0,1)$ sur la droite engendrée par $u=(1,1,1)$.

\section*{Exercice 27.3 -- Gram--Schmidt}
Orthogonaliser $u=(1,1)$ et $v=(1,0)$ puis normaliser.

\section*{Exercice 27.4 -- Moindres carrés}
Ajuster par $y=ax+b$ les points $(0,1),(1,2),(2,2)$ et justifier le caractère minimal.

\section*{Exercice 27.5 -- Polynômes orthogonaux}
Pour $\langle P,Q\rangle=\int_0^1PQ$, trouver la constante $c$ rendant $t-c$ orthogonal à $1$, puis sa norme.

\section*{Exercice 27.6 -- Projection idempotente}
Prouver $p_F(p_F(x))=p_F(x)$ pour une projection orthogonale sur $F$.

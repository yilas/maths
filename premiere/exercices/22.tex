\chapter{Variance, covariance et erreur quadratique}
\section*{Exercice 22.1 -- Moments}
$X$ vaut $0,2,5$ avec probabilités $1/2,1/3,1/6$. Calculer moyenne, variance et écart type.

\section*{Exercice 22.2 -- Changement d'unité}
Si $\mathbb E(X)=10$ et $\operatorname{Var}(X)=9$, donner moyenne et écart type de $Y=2X-5$.

\section*{Exercice 22.3 -- Somme indépendante}
Calculer les moments d'une somme de quatre Bernoulli indépendantes de paramètre $1/4$.

\section*{Exercice 22.4 -- Covariance non nulle}
Pour $Y=2X+1$, exprimer $\operatorname{Cov}(X,Y)$ et $\operatorname{Var}(X+Y)$ en fonction de $\operatorname{Var}(X)$.

\section*{Exercice 22.5 -- Meilleure constante}
Pour une variable de moyenne $3$ et de variance $2$, comparer les erreurs quadratiques des prédictions constantes $0,3,5$.

\section*{Exercice 22.6 -- Décorrélé et dépendant}
$X$ est uniforme sur $\{-1,0,1\}$ et $Y=X^2$. Prouver qu'ils sont décorrélés mais dépendants.
